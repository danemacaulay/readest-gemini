# pdf2epub Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a standalone `pdf2epub` CLI that converts PDFs to reflowable EPUBs via Marker → Pandoc, with auto title/author metadata and a generated text title page.

**Architecture:** New Python package at `~/projects/pdf2epub` (sibling of `reader`, not inside it). Thin CLI orchestrates external `marker_single` + `pandoc`; pure Python modules handle dep checks, PDF/markdown metadata resolution, and subprocess wiring. Calibre plugin is out of scope (Phase 2).

**Tech Stack:** Python 3.11+, `pyproject.toml` / hatchling or setuptools, pytest, `pypdf` (PDF info), `marker-pdf` (pip dep), system `pandoc` + `llama-server`.

**Spec:** `reader/docs/superpowers/specs/2026-10-10-pdf2epub-design.md` (copy into the new repo as `DESIGN.md` in Task 1).

## Global Constraints

- Default Marker mode is `balanced`; `--fast` opts into `fast`.
- Metadata priority: CLI flags > PDF info > markdown heuristics > filename stem.
- Always generate a Pandoc text title page (do not pass `--epub-title-page=false`).
- Do not bundle pandoc, llama.cpp, or model weights.
- `pip install` must pull `marker-pdf`; native missing-dep errors must print brew install hints.
- Do not auto-install system packages.
- Unit tests must not invoke Marker balanced / download models.
- Phase 2 Calibre plugin: do not implement; omit `calibre_plugin/` until Phase 2.
- Repo root for all paths below: `$HOME/projects/pdf2epub` (create if missing).

```python
from __future__ import annotations

import re
from dataclasses import dataclass
from pathlib import Path

from pypdf import PdfReader

_EMPHASIS_RE = re.compile(r"[*_`]+")
_GARBAGE_TITLE_RE = re.compile(
    r"^(untitled|document1|microsoft word\b.*|adobe photoshop\b.*)$",
    re.I,
)


@dataclass(frozen=True)
class BookMeta:
    title: str
    author: str | None


def strip_md_emphasis(text: str) -> str:
    return _EMPHASIS_RE.sub("", text).strip()


def is_garbage_pdf_field(value: str | None) -> bool:
    if value is None:
        return True
    cleaned = value.strip()
    if not cleaned:
        return True
    return bool(_GARBAGE_TITLE_RE.match(cleaned))


def parse_markdown_heuristics(markdown: str) -> BookMeta:
    lines = markdown.splitlines()
    title: str | None = None
    author: str | None = None
    for i, raw in enumerate(lines):
        line = raw.strip()
        if title is None and line.startswith("# "):
            title = strip_md_emphasis(line[2:].strip())
            # scan a few following non-empty lines for author
            for j in range(i + 1, min(i + 6, len(lines))):
                cand = strip_md_emphasis(lines[j].strip())
                if not cand:
                    continue
                if cand.startswith("#"):
                    break
                if _looks_like_author_line(cand):
                    author = _normalize_author_line(cand)
                break
            break
    if not title:
        title = "Untitled"
    return BookMeta(title=title, author=author)


def _looks_like_author_line(line: str) -> bool:
    if len(line) > 180:
        return False
    if line.lower().startswith("abstract"):
        return False
    # names / commas / and / affiliation italics already stripped
    return bool(re.search(r"[A-Za-z].*[A-Za-z]", line)) and (
        "," in line
        or " and " in line.lower()
        or re.search(r"\b(university|institute|laboratory|lab)\b", line, re.I)
        or bool(re.match(r"^[A-Z][a-z]+(?:\s+[A-Z][a-z.\-]+)+", line))
    )


def _normalize_author_line(line: str) -> str:
    # Drop trailing affiliation after common separators if both sides look present
    for sep in ("*", " - ", " – ", " — "):
        if sep in line:
            left = line.split(sep, 1)[0].strip()
            if left and "," in left or " " in left:
                line = left
                break
    # If affiliation glued without separator, keep full line (safer)
    return re.sub(r"\s+", " ", line).strip()


def read_pdf_info(pdf_path: Path) -> BookMeta:
    reader = PdfReader(str(pdf_path))
    info = reader.metadata or {}
    title = info.get("/Title") if info else None
    author = info.get("/Author") if info else None
    # pypdf may return strings already
    title_s = str(title).strip() if title else ""
    author_s = str(author).strip() if author else ""
    return BookMeta(
        title="" if is_garbage_pdf_field(title_s) else title_s,
        author=None if is_garbage_pdf_field(author_s) else author_s,
    )


def resolve_metadata(
    *,
    pdf_path: Path,
    markdown: str,
    title_override: str | None = None,
    author_override: str | None = None,
) -> BookMeta:
    pdf_meta = read_pdf_info(pdf_path)
    md_meta = parse_markdown_heuristics(markdown)

    title = (
        title_override
        or (pdf_meta.title or None)
        or md_meta.title
        or pdf_path.stem
    )
    if title_override:
        title = title_override
    elif pdf_meta.title:
        title = pdf_meta.title
    elif md_meta.title and md_meta.title != "Untitled":
        title = md_meta.title
    else:
        title = pdf_path.stem

    if author_override:
        author: str | None = author_override
    elif pdf_meta.author:
        author = pdf_meta.author
    else:
        author = md_meta.author

    return BookMeta(title=title.strip(), author=author.strip() if author else None)
```

```python
from __future__ import annotations

import shutil
from pathlib import Path


class MissingDependencyError(RuntimeError):
    def __init__(self, name: str, hint: str):
        self.name = name
        self.hint = hint
        super().__init__(f"Missing dependency: {name}\n{hint}")


def which(name: str) -> Path | None:
    found = shutil.which(name)
    return Path(found) if found else None


def require_pandoc() -> Path:
    path = which("pandoc")
    if path is None:
        raise MissingDependencyError("pandoc", "Install with: brew install pandoc")
    return path


def require_marker_single() -> Path:
    path = which("marker_single")
    if path is None:
        raise MissingDependencyError(
            "marker_single",
            "Install this package in an environment with marker-pdf:\n  pip install pdf2epub",
        )
    return path


def require_llama_server() -> Path:
    path = which("llama-server")
    if path is None:
        raise MissingDependencyError(
            "llama-server",
            "Required for balanced mode. Install with: brew install llama.cpp\n"
            "Or pass --fast to skip (lower quality).",
        )
    return path


def check_conversion_deps(*, fast: bool) -> None:
    require_pandoc()
    require_marker_single()
    if not fast:
        require_llama_server()
```

```python
from __future__ import annotations

import shutil
import subprocess
import tempfile
from dataclasses import dataclass
from pathlib import Path

from pdf2epub.deps import (
    check_conversion_deps,
    require_marker_single,
    require_pandoc,
)
from pdf2epub.metadata import BookMeta, resolve_metadata


@dataclass(frozen=True)
class ConvertOptions:
    pdf_path: Path
    out_path: Path | None = None
    title: str | None = None
    author: str | None = None
    fast: bool = False
    keep: bool = False
    work_dir: Path | None = None


def default_out_path(pdf_path: Path) -> Path:
    return pdf_path.with_suffix(".epub")


def find_marker_markdown(marker_out_dir: Path, pdf_stem: str) -> Path:
    candidate = marker_out_dir / pdf_stem / f"{pdf_stem}.md"
    if candidate.is_file():
        return candidate
    matches = list(marker_out_dir.rglob("*.md"))
    if len(matches) == 1:
        return matches[0]
    raise FileNotFoundError(
        f"Marker markdown not found under {marker_out_dir} (expected {candidate})"
    )


def build_pandoc_cmd(
    *,
    pandoc: Path,
    md_path: Path,
    epub_path: Path,
    meta: BookMeta,
) -> list[str]:
    cmd = [
        str(pandoc),
        str(md_path),
        "-o",
        str(epub_path),
        "--toc",
        "--toc-depth=3",
        f"--resource-path={md_path.parent}",
        f"--metadata=title:{meta.title}",
    ]
    if meta.author:
        cmd.append(f"--metadata=author:{meta.author}")
    return cmd


def convert_pdf(opts: ConvertOptions) -> Path:
    pdf_path = opts.pdf_path.expanduser().resolve()
    if not pdf_path.is_file():
        raise FileNotFoundError(f"PDF not found: {pdf_path}")

    check_conversion_deps(fast=opts.fast)
    marker = require_marker_single()
    pandoc = require_pandoc()
    out_path = (opts.out_path or default_out_path(pdf_path)).expanduser().resolve()

    def _run(work: Path) -> Path:
        work.mkdir(parents=True, exist_ok=True)
        mode = "fast" if opts.fast else "balanced"
        marker_cmd = [
            str(marker),
            str(pdf_path),
            "--output_dir",
            str(work),
            "--output_format",
            "markdown",
            "--mode",
            mode,
        ]
        proc = subprocess.run(marker_cmd, capture_output=True, text=True)
        if proc.returncode != 0:
            raise RuntimeError(
                f"marker_single failed ({proc.returncode}):\n{proc.stderr or proc.stdout}"
            )

        md_path = find_marker_markdown(work, pdf_path.stem)
        markdown = md_path.read_text(encoding="utf-8")
        meta = resolve_metadata(
            pdf_path=pdf_path,
            markdown=markdown,
            title_override=opts.title,
            author_override=opts.author,
        )
        out_path.parent.mkdir(parents=True, exist_ok=True)
        pandoc_cmd = build_pandoc_cmd(
            pandoc=pandoc, md_path=md_path, epub_path=out_path, meta=meta
        )
        proc2 = subprocess.run(pandoc_cmd, capture_output=True, text=True)
        if proc2.returncode != 0:
            raise RuntimeError(
                f"pandoc failed ({proc2.returncode}):\n{proc2.stderr or proc2.stdout}"
            )
        if not out_path.is_file():
            raise RuntimeError(f"pandoc reported success but EPUB missing: {out_path}")
        return out_path

    if opts.keep:
        keep_root = opts.work_dir or (pdf_path.parent / f"{pdf_path.stem}.pdf2epub")
        return _run(keep_root)

    if opts.work_dir is not None:
        return _run(opts.work_dir)

    with tempfile.TemporaryDirectory(prefix="pdf2epub-") as tmp:
        return _run(Path(tmp))
```

Note: when `keep=True`, Marker output stays under `{stem}.pdf2epub/`. When tests pass `work_dir` without `keep`, use that dir (no delete). Remove unused `shutil` import if not needed — if `keep` copies are not required, drop `shutil`.

```python
from __future__ import annotations

import argparse
import sys
from pathlib import Path

from pdf2epub.convert import ConvertOptions, convert_pdf
from pdf2epub.deps import MissingDependencyError


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(
        prog="pdf2epub",
        description="Convert a PDF to a reflowable EPUB via Marker + Pandoc.",
    )
    p.add_argument("pdf", type=Path, help="Input PDF path")
    p.add_argument("--out", type=Path, default=None, help="Output EPUB path")
    p.add_argument("--title", default=None, help="Override title")
    p.add_argument("--author", default=None, help="Override author")
    p.add_argument(
        "--fast",
        action="store_true",
        help="Use Marker fast mode (skip llama-server; lower quality)",
    )
    p.add_argument(
        "--keep",
        action="store_true",
        help="Keep intermediate Marker markdown/images beside the PDF",
    )
    return p


def main(argv: list[str] | None = None) -> int:
    parser = build_parser()
    try:
        args = parser.parse_args(argv)
    except SystemExit as e:
        return int(e.code) if e.code is not None else 2

    opts = ConvertOptions(
        pdf_path=args.pdf,
        out_path=args.out,
        title=args.title,
        author=args.author,
        fast=args.fast,
        keep=args.keep,
    )
    try:
        out = convert_pdf(opts)
    except MissingDependencyError as e:
        print(str(e), file=sys.stderr)
        return 2
    except FileNotFoundError as e:
        print(str(e), file=sys.stderr)
        return 1
    except Exception as e:
        print(f"Conversion failed: {e}", file=sys.stderr)
        return 1
    print(out)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
